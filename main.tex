% define some basic colours. FOUND BY TESTING 2026-09: this file used
% to \newcommand these while pstricks (loaded above, since removed) was
% also loaded and already defined them, giving a hard "Command \blue
% already defined" error. Now that pstricks is gone, plain \newcommand
% is correct again -- and, since document.tex's own prose uses both
% {\red Define} (no braced argument right after) and \bs{\blue{alt}}
% (an immediately-following braced group), these stay deliberately
% declarative (no argument) rather than argument-taking: {\blue{alt}}
% is just \blue switching color, then a separate group {alt} inheriting
% it -- an argument-taking \blue{#1} would have broken the first style
% instead (silently swallowing the next single token as its argument,
% e.g. {\red Define} -> "D", leaving "efine" uncolored).
\setbeamercolor{blue}{fg=blue!90}
\newcommand{\blue}{\usebeamercolor[fg]{blue}}
\setbeamercolor{red}{fg=red!80,bg=yellow}
\newcommand{\red}{\usebeamercolor[fg]{red}}
\setbeamercolor{black}{fg=black!99}
\newcommand{\black}{\usebeamercolor[fg]{black}}
\setbeamercolor{green}{fg=green!80}
\newcommand{\green}{\usebeamercolor[fg]{green}}

% \bs -- a literal backslash, for showing command names as text (e.g.
% \bs documentclass prints "\documentclass"). Used throughout
% document.tex but was never actually defined anywhere in this
% template -- FOUND BY TESTING 2026-09, another pre-existing gap.
\newcommand{\bs}{\textbackslash}
